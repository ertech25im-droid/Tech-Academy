        /**
         * Validate email format using regular expression
         * @param {string} email - The email address to validate
         * @returns {boolean} - True if email is valid, false otherwise
         */
        function isValidEmail(email) {
            // Regular expression pattern for email validation
            const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            return emailPattern.test(email);
        }

        /**
         * Validate individual form field
         * @param {string} fieldId - The ID of the field to validate

        function validateField(fieldId) {
            const field = document.getElementById(fieldId);
            const value = field.value.trim();
            const errorId = fieldId + 'Error';

            if (fieldId === 'name') {
                if (value === '') {
                    showError(errorId, 'Please enter your full name');
                } else if (value.length < 3) {
                    showError(errorId, 'Name must be at least 3 characters long');
                } else {
                    clearFieldError(errorId);
                }
            } else if (fieldId === 'email') {
                if (value === '') {
                    showError(errorId, 'Please enter your email address');

                    showError(errorId, 'Please enter a valid email address');
                } else {
                    clearFieldError(errorId);
                }
            } else if (fieldId === 'message') {
                if (value === '') {
                    showError(errorId, 'Please enter your message');
                } else if (value.length < 10) {
                    showError(errorId, 'Message must be at least 10 characters long');
                } else {
                    clearFieldError(errorId);
                }
            }
        }
